Recommendation: Use FRED as your primary API. It aggregates BLS data (plus hundreds of other sources) into a single, cleaner API with better documentation, more flexible queries, and no CORS issues. Use BLS directly only if you need data the same morning it's released (FRED has a ~30 minute delay from BLS releases).

Step 3 — The Integration Code

Below is the complete API module you'll paste into your index.html. It replaces the static DATA object with live FRED calls.

3A — FRED API Fetch Function

This is the core function. It calls FRED, gets JSON back, and returns an array of {date, value} objects:

// ═══════════════════════════════════════════════════
// FRED API MODULE — paste this ABOVE your DATA object
// ═══════════════════════════════════════════════════

const FRED_API_KEY = localStorage.getItem('fred_api_key') || 'YOUR_KEY_HERE';
const FRED_BASE   = 'https://api.stlouisfed.org/fred/series/observations';

/**
 * Fetch a FRED series and return parsed observations.
 *
 * @param {string} seriesId   — e.g. 'CPIAUCSL'
 * @param {string} startDate  — e.g. '2020-01-01'
 * @param {string} units      — 'lin' (level), 'pc1' (YoY %), 'pch' (MoM %)
 * @param {string} freq       — 'q' (quarterly), 'a' (annual), 'm' (monthly)
 * @returns {Promise<Array<{date: string, value: number}>>}
 */
async function fetchFRED(seriesId, startDate = '2020-01-01', units = 'pc1', freq = 'm') {
  const url = `${FRED_BASE}?series_id=${seriesId}`
    + `&api_key=${FRED_API_KEY}`
    + `&file_type=json`
    + `&observation_start=${startDate}`
    + `&units=${units}`
    + `&frequency=${freq}`
    + `&sort_order=asc`;

  try {
    const res = await fetch(url);
    const json = await res.json();
    return json.observations
      .filter(o => o.value !== '.')  // FRED uses '.' for missing
      .map(o => ({
        date:  o.date,
        value: parseFloat(o.value)
      }));
  } catch (err) {
    console.error(`FRED fetch failed for ${seriesId}:`, err);
    return [];
  }
}

3B — The "units" Parameter (This is the Key Concept)

FRED can do math for you. Instead of fetching raw index levels and calculating YoY change yourself, you tell FRED what transform you want:

3C — Loading All Dashboard Data

// ═══════════════════════════════════════════════════
// LOAD ALL SERIES — call this on page load
// ═══════════════════════════════════════════════════

async function loadAllData() {
  const start = '2020-01-01';

  // Fetch everything in parallel for speed
  const [
    cpiYoY_q,      // CPI YoY%, quarterly
    cpiYoY_a,      // CPI YoY%, annual
    cpiMoM,        // CPI MoM%
    coreCpiYoY_q,  // Core CPI YoY%, quarterly
    ppiYoY_q,      // PPI YoY%, quarterly
    ppiYoY_a,      // PPI YoY%, annual
    ppiMoM,        // PPI MoM%
    corePpiYoY_q,  // Core PPI YoY%, quarterly
    nfpLevel_q,    // Nonfarm payrolls, quarterly total
    nfpChange,     // NFP month-over-month change
    unrate_q,      // Unemployment rate, quarterly
    aheYoY_q,      // Avg Hourly Earnings YoY%
    jolts_q,       // JOLTS openings, quarterly
    u6rate,        // U-6 rate
    lfpr,          // Labor force participation
  ] = await Promise.all([
    fetchFRED('CPIAUCSL',  start, 'pc1', 'q'),
    fetchFRED('CPIAUCSL',  start, 'pc1', 'a'),
    fetchFRED('CPIAUCSL',  start, 'pch', 'm'),
    fetchFRED('CPILFESL',  start, 'pc1', 'q'),
    fetchFRED('PPIFIS',    start, 'pc1', 'q'),
    fetchFRED('PPIFIS',    start, 'pc1', 'a'),
    fetchFRED('PPIFIS',    start, 'pch', 'm'),
    fetchFRED('PPIFES',    start, 'pc1', 'q'),
    fetchFRED('PAYEMS',    start, 'chg', 'q'),
    fetchFRED('PAYEMS',    start, 'chg', 'm'),
    fetchFRED('UNRATE',    start, 'lin', 'q'),
    fetchFRED('CES0500000003', start, 'pc1', 'q'),
    fetchFRED('JTSJOL',    start, 'lin', 'q'),
    fetchFRED('U6RATE',    start, 'lin', 'm'),
    fetchFRED('CIVPART',   start, 'lin', 'm'),
  ]);

  // ── Convert FRED arrays into your DATA object format ──
  // Helper: converts [{date, value}] to {"2024-Q1": 3.2, ...}
  function toQuarterlyMap(arr) {
    const map = {};
    arr.forEach(({ date, value }) => {
      const d = new Date(date);
      const q = Math.ceil((d.getMonth() + 1) / 3);
      map[`${d.getFullYear()}-Q${q}`] = Math.round(value * 10) / 10;
    });
    return map;
  }

  // Helper: converts [{date, value}] to {2020: 1.2, 2021: 4.7, ...}
  function toAnnualMap(arr) {
    const map = {};
    arr.forEach(({ date, value }) => {
      map[new Date(date).getFullYear()] = Math.round(value * 10) / 10;
    });
    return map;
  }

  // Now populate your DATA object:
  DATA.cpi.quarterly     = toQuarterlyMap(cpiYoY_q);
  DATA.cpi.annual        = toAnnualMap(cpiYoY_a);
  DATA.cpiCore.quarterly = toQuarterlyMap(coreCpiYoY_q);
  DATA.ppi.quarterly     = toQuarterlyMap(ppiYoY_q);
  DATA.ppi.annual        = toAnnualMap(ppiYoY_a);
  DATA.ppiCore.quarterly = toQuarterlyMap(corePpiYoY_q);
  DATA.nfp.quarterly     = toQuarterlyMap(nfpLevel_q);
  DATA.unrate.quarterly  = toQuarterlyMap(unrate_q);
  DATA.ahe.quarterly     = toQuarterlyMap(aheYoY_q);
  DATA.jolts.quarterly   = toQuarterlyMap(jolts_q);

  // Latest values from the most recent monthly observations
  const last = arr => arr[arr.length - 1]?.value ?? null;
  const prev = arr => arr[arr.length - 2]?.value ?? null;

  DATA.cpi.latest     = last(cpiYoY_q);
  DATA.cpi.prevMonth  = prev(cpiYoY_q);
  DATA.cpi.mom        = last(cpiMoM);
  DATA.ppi.latest     = last(ppiYoY_q);
  DATA.ppi.prevMonth  = prev(ppiYoY_q);
  DATA.ppi.mom        = last(ppiMoM);
  DATA.unrate.latest  = last(unrate_q);
  DATA.unrate.u6      = last(u6rate);
  DATA.unrate.lfpr    = last(lfpr);
  DATA.nfp.latestMonth = last(nfpChange);
  DATA.nfp.prevMonth   = prev(nfpChange);

  // Re-render the dashboard with live data
  buildSummaryCards();
  buildTicker();
  renderView(currentView);

  console.log('✓ Live FRED data loaded', DATA);
}

// ── Call on page load ──
loadAllData();

3D — Where Exactly to Paste It

Open your index.html and find the <script> tag near the bottom. The structure should be:

// 1. Keep your existing DATA object as the FALLBACK
//    (it loads instantly while APIs are fetching)

const DATA = { ... };  // ← your existing static data stays here

// 2. Paste the FRED API MODULE here (fetchFRED function)

// 3. Paste loadAllData() here

// 4. Everything else (chart functions, renderers, etc.) stays the same

// 5. At the very bottom, AFTER renderView('overview'), add:
loadAllData();  // ← this overwrites static data with live data
Why keep the static data? It acts as a fallback. The page renders instantly with static numbers, then FRED data loads in 1-2 seconds and overwrites them. The user sees no blank screen or loading spinner.

Step 5 — If You Need BLS Direct (Advanced)

If you need same-morning BLS data before FRED updates, you'll need a small proxy because BLS blocks browser requests (CORS). Two options:

Option A: Cloudflare Worker (Free, 5 Min Setup)

// Deploy this as a Cloudflare Worker at workers.dev
export default {
  async fetch(request) {
    const url = new URL(request.url);
    const blsUrl = 'https://api.bls.gov/publicAPI/v2/timeseries/data/';

    const body = await request.json();
    const resp = await fetch(blsUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body)
    });

    const data = await resp.json();
    return new Response(JSON.stringify(data), {
      headers: {
        'Content-Type': 'application/json',
        'Access-Control-Allow-Origin': '*',
      }
    });
  }
};

Option B: Simple Node.js Proxy (If You Prefer Local)

// server.js — run with: node server.js
const express = require('express');
const fetch = require('node-fetch');
const app = express();
app.use(express.json());
app.use(express.static('.'));  // serves your index.html

app.post('/api/bls', async (req, res) => {
  const resp = await fetch(
    'https://api.bls.gov/publicAPI/v2/timeseries/data/',
    { method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(req.body) }
  );
  res.json(await resp.json());
});

app.listen(3000, () => console.log('Proxy on http://localhost:3000'));

Security Note on API Keys

API keys in frontend code are visible to anyone. For these specific APIs (FRED, BLS, BEA), this is acceptable because they are free government APIs with no billing. But follow these rules:

• Never use this pattern for paid APIs (Finnhub, Alpha Vantage with premium, etc.)
• The localStorage approach keeps your key out of your GitHub repository
• If deploying publicly, consider a Cloudflare Worker or Vercel serverless function to hide the key
• For Ghost Strategies client tools, always use environment variables in a backend
